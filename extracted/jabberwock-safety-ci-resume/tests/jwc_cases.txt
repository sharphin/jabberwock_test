enum Shape {
    Circle(f64 r);
    Rect(f64 w, f64 h);

    fn take Shape scaled(f64 k) {
        match this {
            case Circle(r) -> Shape.Circle(r * k);
            case Rect(w, h) -> Shape.Rect(w * k, h * k);
        }
    }
    fn f64 area() { match this { case Circle(r) -> 3.0 * r * r; case Rect(w, h) -> w * h; } }
}

enum Slot {
    Full(own i64 v);
    Empty;

    fn take i64 open() { match this { case Full(v) -> *v; case Empty -> 0; } }
}

fn main() {
    let t = Tag { name: "a", n: 1 };

    let c = Shape.Circle(1.0);
    let d = c.scaled(2.0);
    print(d.area(), c.area());
    let s = Slot.Full(own 7);
    print(s.open(), Slot.Empty.open());
}

// 並列の単語数え上げ: 行を worker に配り、結果の辞書を chan で集める
fn i64[str] count(str[] lines) {
    let i64[str] mut out = [];
    for line in lines {
        for raw in line.split(" ") {
            let w = raw.trim().to_lower();
            if w == "" { continue; }
            if out.contains(w) { out[w] += 1; } else { out[w] = 1; }
        }
    }
    out
}

fn main() {
    let text = """
the quick brown fox
jumps over the lazy dog
the dog sleeps
a fox is quick""";
    let lines = text.lines();
    let workers = 3;
    let (str[])[] mut parts = [];
    for i in range(workers) { parts.push([]); }
    for (i, l) in lines {
        parts[i % workers].push(l);
    }
    let chan<i64[str]> results = chan();
    let task<()>[] mut ts = [];
    for p in parts.drain() {
        // drain で取り出した要素は値なので、スレッドへ渡せる
        ts.push(spawn(() -> { results.send(count(p)); }));
    }
    let i64[str] mut total = [];
    for i in range(workers) {
        match results.recv() {
            case Some(m) -> {
                for (k, v) in m {
                    if total.contains(k) { total[k] += v; } else { total[k] = v; }
                }
            }
            case None -> {}
        }
    }
    for t in ts { t.join(); }
    let mut keys = total.keys();
    // 引数に書いたクロージャは total を借りるだけ（ムーブしない）
    keys.sort_by((a, b) -> total[a] > total[b] || (total[a] == total[b] && a < b));
    for k in keys {
        print("{k}: {total[k]}");
    }
}

struct Job { i64 id; own i64 data; }

fn main() {
    let chan<Job> jobs = chan();
    let chan<own i64> done = chan();
    let w = spawn(() -> {
        loop {
            match jobs.recv() {
                case Some(j) -> done.send(own (*j.data * 10 + j.id));
                case None -> break;
            }
        }
        done.close();
    });
    for i in range(3) {
        jobs.send(Job { id: i, data: own i });
    }
    jobs.close();
    let mut total = 0;
    loop {
        match done.recv() {
            case Some(x) -> { total += *x; }
            case None -> break;
        }
    }
    w.join();
    print(total);
}

struct Account {
    str name;
    own i64 balance;

    fn mut Result<i64, str> withdraw(i64 n) {
        if n > *balance {
            return Err("残高が足りません");
        }
        *balance = *balance - n;
        Ok(*balance)
    }
}

fn Result<i64, str> twice(ref mut Account a, i64 n) {
    a.withdraw(n)?;
    a.withdraw(n)
}

fn main() {
    let mut a = own Account { name: "x", balance: own 100 };
    print(a.withdraw(30));
    print(twice(ref mut a, 30));
    print(twice(ref mut a, 30));
}

// 二分探索木（試しに書いたプログラム）。ref の引数には ref を書かずに渡せる（自動で借りる）
struct Node {
    i64 key;
    Option<own Node> left;
    Option<own Node> right;
}

struct Tree {
    Option<own Node> root;
    i64 size;

    fn mut insert(i64 key) {
        size += 1;
        insert_at(ref mut root, key);
    }

    fn bool contains(i64 key) {
        let mut cur = ref root;
        loop {
            match cur {
                case None -> return false;
                case Some(n) -> {
                    if n.key == key { return true; }
                    if key < n.key { cur = ref n.left; } else { cur = ref n.right; }
                }
            }
        }
    }
}

fn insert_at(ref mut Option<own Node> slot, i64 key) {
    match slot {
        case None -> *slot = Some(own Node { key: key, left: None, right: None });
        case Some(n) -> {
            if key < n.key { insert_at(ref mut n.left, key); } else { insert_at(ref mut n.right, key); }
        }
    }
}

fn walk(ref Option<own Node> t, ref mut i64[] out) {
    match t {
        case None -> {}
        case Some(n) -> {
            walk(n.left, out);
            out.push(n.key);
            walk(n.right, out);
        }
    }
}

fn main() {
    let mut t = Tree { root: None, size: 0 };
    for k in [5, 3, 8, 1, 4, 9, 7] {
        t.insert(k);
    }
    let i64[] mut keys = [];
    walk(t.root, ref mut keys);
    print("size {t.size}: {keys}");
    print(t.contains(4), t.contains(6));
}

// 在庫と注文の処理: 構造体の辞書・interface・クロージャのリスト・並べ替え
struct Item {
    str name;
    i64 price;
    i64 stock;
}

struct Order {
    i64 id;
    str item;
    i64 qty;
}

interface Discount {
    fn i64 apply(ref Order o, i64 total);
    fn str label();
}

struct Bulk {
    i64 min_qty;
    i64 percent;
    fn i64 apply(ref Order o, i64 total) {
        if o.qty >= min_qty { total - total * percent / 100 } else { total }
    }
    fn str label() { "まとめ買い {percent}%" }
}

struct Coupon {
    i64 off;
    fn i64 apply(ref Order o, i64 total) {
        if total > off { total - off } else { 0 }
    }
    fn str label() { "クーポン {off} 円" }
}

struct Shop {
    Item[str] items;
    (own Discount)[] discounts;
    str[] log;

    fn mut Result<i64, str> sell(ref Order o) {
        if !items.contains(o.item) { return Err("{o.item} は扱っていません"); }
        let stock = items[o.item].stock;
        if stock < o.qty { return Err("{o.item} の在庫が足りません（残り {stock}）"); }
        items[o.item].stock -= o.qty;
        let mut total = items[o.item].price * o.qty;
        for d in discounts {
            let before = total;
            total = d.apply(o, total);
            if total != before { log.push("#{o.id}: {d.label()} で {before} → {total}"); }
        }
        Ok(total)
    }
}

fn main() {
    let mut shop = Shop { items: [], discounts: [], log: [] };
    shop.items["りんご"] = Item { name: "りんご", price: 120, stock: 10 };
    shop.items["みかん"] = Item { name: "みかん", price: 80, stock: 3 };
    shop.discounts.push(own Bulk { min_qty: 5, percent: 10 });
    shop.discounts.push(own Coupon { off: 50 });

    let Order[] mut orders = [
        Order { id: 1, item: "りんご", qty: 6 },
        Order { id: 2, item: "みかん", qty: 5 },
        Order { id: 3, item: "ぶどう", qty: 1 },
        Order { id: 4, item: "みかん", qty: 2 },
    ];
    orders.sort_by((a, b) -> a.qty < b.qty);
    let mut sales = 0;
    for o in orders {
        match shop.sell(o) {
            case Ok(t) -> { sales += t; print("#{o.id} {o.item} x{o.qty}: {t} 円"); }
            case Err(m) -> print("#{o.id}: {m}");
        }
    }
    print("売上 {sales} 円");
    for l in shop.log { print(l); }
    let str[] mut low = [];
    for (n, it) in shop.items {
        if it.stock < 5 { low.push(n); }
    }
    print("在庫が少ない: {low}");
}

interface Unit {
    fn str name();
    fn i64 hp();
    fn mut hit(i64 dmg);
}

struct Slime {
    i64 life;
    fn str name() { "スライム" }
    fn i64 hp() { life }
    fn mut hit(i64 dmg) { life -= dmg; }
}

struct Dragon {
    i64 life;
    i64 armor;
    fn str name() { "ドラゴン" }
    fn i64 hp() { life }
    fn mut hit(i64 dmg) {
        if dmg > armor { life -= dmg - armor; }
    }
}

struct Hero {
    str name;
    i64 power;
    Option<ref mut Unit> target;
}

fn main() {
    let (own Unit)[] mut enemies = [own Slime { life: 10 }, own Dragon { life: 30, armor: 2 }];
    let mut hero = Hero { name: "勇者", power: 7, target: None };
    let mut turn = 0;
    while enemies.len() > 0 {
        turn += 1;
        // 先頭の敵を狙う
        hero.target = Some(ref mut enemies[0]);
        match hero.target {
            case Some(t) -> {
                t.hit(hero.power);
                print("{turn}: {t.name()} の残り {t.hp()}");
                if t.hp() <= 0 {
                    print("{t.name()} を倒した");
                    enemies.remove(0);
                }
            }
            case None -> {}
        }
        if turn > 20 { break; }
    }
}

// 単語の数え上げ（試しに書いたプログラム）。辞書は入れた順に回る
// ファイル名のあとに文章を渡すと、それを数える（jabberwock words.jw "a b a"）
fn i64[str] count_words(ref str[] words) {
    let i64[str] mut out = [];
    for w in words {
        if out.contains(w) {
            out[w] += 1;
        } else {
            out[w] = 1;
        }
    }
    out
}

fn main() {
    let mut text = "the cat and the dog and the bird";
    let a = args();
    if a.len() > 0 {
        text = a[0];
    }
    let words = text.split(" ");
    let counts = count_words(words);
    for (w, n) in counts {
        print("{w}: {n}");
    }
}

// 電卓: 字句解析 → 構文木（再帰する enum）→ 計算。エラーは Result と ?。match は形を重ねて書ける（`Some(Op("-"))`）
enum Tok {
    Num(i64 n);
    Op(str op);
    LParen;
    RParen;
}

enum Expr {
    Num(i64 n);
    Bin(str op, own Expr l, own Expr r);
    Neg(own Expr e);
}

fn Result<Tok[], str> lex(str src) {
    let Tok[] mut out = [];
    let cs = src.chars();
    let mut i = 0;
    while i < cs.len() {
        let c = cs[i];
        if c == " " { i += 1; continue; }
        if "0123456789".contains(c) {
            let mut n = 0;
            while i < cs.len() && "0123456789".contains(cs[i]) {
                n = n * 10 + ok_or(cs[i].to_int(), "数ではない")?;
                i += 1;
            }
            out.push(Tok.Num(n));
            continue;
        }
        match c {
            case "(" -> out.push(Tok.LParen);
            case ")" -> out.push(Tok.RParen);
            case "+", "-", "*", "/" -> out.push(Tok.Op(c));
            default -> return Err("知らない文字 '{c}'（{i} 文字目）");
        }
        i += 1;
    }
    Ok(out)
}

struct Parser {
    Tok[] toks;
    i64 pos;

    fn Option<Tok> peek() {
        if pos < toks.len() { Some(toks[pos]) } else { None }
    }

    fn mut Result<own Expr, str> expr() {
        let mut l = term()?;
        loop {
            match peek() {
                case Some(Op(op)) -> {
                    if op != "+" && op != "-" { break; }
                    pos += 1;
                    let r = term()?;
                    l = own Expr.Bin(op, l, r);
                }
                default -> break;
            }
        }
        Ok(l)
    }

    fn mut Result<own Expr, str> term() {
        let mut l = atom()?;
        loop {
            match peek() {
                case Some(Op(op)) -> {
                    if op != "*" && op != "/" { break; }
                    pos += 1;
                    let r = atom()?;
                    l = own Expr.Bin(op, l, r);
                }
                default -> break;
            }
        }
        Ok(l)
    }

    fn mut Result<own Expr, str> atom() {
        let t = peek();
        pos += 1;
        match t {
            case Some(Num(n)) -> Ok(own Expr.Num(n));
            case Some(Op("-")) -> Ok(own Expr.Neg(atom()?));
            case Some(Op(op)) -> Err("ここに {op} は書けません");
            case Some(LParen) -> {
                let e = expr()?;
                match peek() {
                    case Some(RParen) -> { pos += 1; Ok(e) }
                    default -> Err("')' がありません");
                }
            }
            case Some(RParen) -> Err("')' が多すぎます");
            case None -> Err("式が途中で終わっています");
        }
    }
}

fn Result<i64, str> eval(ref Expr e) {
    match e {
        case Num(n) -> Ok(n);
        case Neg(x) -> Ok(-eval(x)?);
        case Bin(op, l, r) -> {
            let a = eval(l)?;
            let b = eval(r)?;
            match op {
                case "+" -> Ok(a + b);
                case "-" -> Ok(a - b);
                case "*" -> Ok(a * b);
                default -> if b == 0 { Err("0 で割りました") } else { Ok(a / b) }
            }
        }
    }
}

fn Result<i64, str> run(str src) {
    let mut p = Parser { toks: lex(src)?, pos: 0 };
    let e = p.expr()?;
    if p.pos < p.toks.len() { return Err("余分なものがあります"); }
    eval(e)
}

fn main() {
    for src in ["1 + 2 * 3", "(1 + 2) * 3", "-4 + 10 / 2", "2 * (3 + ", "8 / (4 - 4)", "7 % 2"] {
        match run(src) {
            case Ok(v) -> print("{src} = {v}");
            case Err(m) -> print("{src}: エラー: {m}");
        }
    }
}

struct Node { i64 v; Option<own Node> next; }
enum Slot { Empty; Full(own i64 v); }
fn i64 sum(ref Option<own Node> l) {
    let mut s = 0;
    let mut cur = l;
    loop {
        match cur {
            case None -> return s;
            case Some(n) -> { s += n.v; cur = ref n.next; }
        }
    }
}
fn Option<own Node> build(i64 n) {
    let Option<own Node> mut head = None;
    for i in range(n) { head = Some(own Node { v: i, next: head }); }
    head
}

enum Shape {
    Circle(f64 r);
    Rect(f64 w, f64 h);

    fn f64 area() {
        match this {
            case Circle(r) -> 3.0 * r * r;
            case Rect(w, h) -> w * h;
        }
    }
    fn mut grow() {
        match this {
            case Circle(r) -> *this = Shape.Circle(r * 2.0);
            case Rect(w, h) -> *this = Shape.Rect(w * 2.0, h * 2.0);
        }
    }
    static fn Shape square(f64 s) { Shape.Rect(s, s) }
    fn bool big() { area() > 100.0 }
}

enum Tree<T> {
    Leaf(T v);
    Node(own Tree<T> l, own Tree<T> r);

    fn i64 count() {
        match this {
            case Leaf(_) -> 1;
            case Node(l, r) -> l.count() + r.count();
        }
    }
}

fn main() {
    let mut s = Shape.square(3.0);
    print(s.area());
    s.grow();
    print(s.area(), s.big());
    let o = own Shape.Circle(10.0);
    print(o.big(), Shape.Rect(1.0, 2.0).area());
    let mut xs = [Shape.Circle(1.0), Shape.Circle(1.0)];
    xs[1].grow();
    print(xs[0].area(), xs[1].area());
    let t = Tree.Node(own Tree.Leaf("a"), own Tree.Node(own Tree.Leaf("b"), own Tree.Leaf("c")));
    print(t.count());
}

enum Tok { Num(i64 n); Op(str op); End; }
enum E { Lit(i64 n); Add(own E l, own E r); Neg(own E e); }
struct Box2 { own i64 v; }

fn str show(Option<Tok> t) {
    match t {
        case Some(Num(0)) -> "zero";
        case Some(Num(n)) -> "num {n}";
        case Some(Op("+")) -> "plus";
        case Some(Op(o)) -> "op {o}";
        case Some(End), None -> "end";
    }
}

fn i64 simp(ref E e) {
    match e {
        case Add(Lit(0), r) -> simp(r);
        case Add(l, Lit(0)) -> simp(l);
        case Add(l, r) -> simp(l) + simp(r);
        case Neg(Neg(x)) -> simp(x);
        case Neg(x) -> -simp(x);
        case Lit(n) -> n;
    }
}

fn main() {
    print(show(Some(Tok.Num(0))), show(Some(Tok.Num(5))), show(Some(Tok.Op("+"))), show(Some(Tok.Op("*"))), show(None), show(Some(Tok.End)));
    let e = own E.Add(own E.Lit(0), own E.Neg(own E.Neg(own E.Add(own E.Lit(3), own E.Lit(0)))));
    print(simp(e));
    let Option<Option<own i64>> x = Some(Some(own 4));
    match x {
        case Some(Some(v)) -> print(*v);
        case Some(None) -> print("inner none");
        case None -> print("none");
    }
    let Result<Option<Box2>, str> y = Ok(Some(Box2 { v: own 9 }));
    match ref y {
        case Ok(Some(_)) -> print("ref");
        default -> print("other");
    }
    match y {
        case Ok(Some(b)) -> print(*b.v);
        case Ok(None) -> {}
        case Err(_) -> {}
    }
    let Option<bool> z = Some(true);
    match z { case Some(true) -> print("t"); case Some(false) -> print("f"); case None -> {} }
}

    let mut v = [E { k: 3, i: 0 }, E { k: 1, i: 1 }, E { k: 3, i: 2 }, E { k: 2, i: 3 }];
    v.sort_by((x, y) -> { *cr = *cr + 1; x.k < y.k });
    v.sort_unstable_by((x, y) -> { *cr = *cr + 1; x.i > y.i });
    let mut ns = [5, 2, 9, 1];
    ns.sort_unstable();
    let mut a = [4.5, -1.0, 2.0];
    a.sort_unstable();
    a.sort_unstable_by((x, y) -> x > y);
    for e in v { print(e.k, e.i); }
    print(ns, a, calls);
}
